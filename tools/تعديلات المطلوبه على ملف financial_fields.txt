// جلب الكيانات مع حساباتها (للقوائم المنسدلة)
function get_accounts_list($pdo, $parent_account_code) {
    $stmt_parent = $pdo->prepare("SELECT id FROM unified_accounts WHERE account_code = ?");
    $stmt_parent->execute([$parent_account_code]);
    $parent_id = $stmt_parent->fetchColumn();
    if (!$parent_id) return [];
    
    $stmt = $pdo->prepare("
        SELECT ua.id, ua.account_code, ua.account_name_ar,
            (SELECT id FROM customers WHERE account_id = ua.id LIMIT 1) as customer_id,
            (SELECT id FROM agents WHERE account_id = ua.id LIMIT 1) as agent_id,
            (SELECT id FROM suppliers WHERE account_id = ua.id LIMIT 1) as supplier_id
        FROM unified_accounts ua
        WHERE ua.parent_id = ? AND ua.account_status = 'active'
        ORDER BY ua.account_code ASC
    ");
    $stmt->execute([$parent_id]);
    $accounts = [];
    while ($row = $stmt->fetch()) {
        $row['display_name'] = $row['account_code'] . ' - ' . $row['account_name_ar'];
        $row['name'] = $row['account_name_ar'];
        $accounts[] = $row;
    }
    return $accounts;
}

                        'data-sell': curr.exchange_rate_sell ?? 1,
                        'data-rate': curr.exchange_rate ?? 1,
                        selected: isSelected
                    }).text(curr.currency_name));
                });
                updateLogic();
            });
            return;
        }
        $.get(window.location.pathname, { action: 'get_active_currencies', account_id: accountId }, function(currencies) {
            const select = $('#' + currencySelectId);
            const currentValue = select.val();
            select.empty();
            if (!currencies || currencies.length === 0) {
                $.get(window.location.pathname, { action: 'get_active_currencies', account_id: 'all' }, function(response) {
                    const allCurrencies = JSON.parse(response);
                    allCurrencies.forEach(curr => {
                        const isSelected = curr.is_default ? 'selected' : '';
                        select.append($('<option>', {
                            value: curr.id,
                            'data-symbol': curr.currency_symbol,
                            'data-buy': curr.exchange_rate_buy ?? 1,
                            'data-sell': curr.exchange_rate_sell ?? 1,
                            'data-rate': curr.exchange_rate ?? 1,
                            selected: isSelected
                        }).text(curr.currency_name));
                    });
                    updateLogic();
                });
                return;
            }
            currencies.forEach(curr => {
                const isSelected = (curr.id == currentValue) || curr.is_default ? 'selected' : '';
                select.append($('<option>', {
                    value: curr.id,
                    'data-symbol': curr.currency_symbol,
                    'data-buy': curr.exchange_rate_buy ?? 1,
                    'data-sell': curr.exchange_rate_sell ?? 1,
                    'data-rate': curr.exchange_rate ?? 1,
                    selected: isSelected
                }).text(curr.currency_name));
            });
            updateLogic();
        }, 'json');
    }

    // دالة معالجة نوع التوصيل
    function handleDeliveryType(type, selectId, labelId, receivedFieldId) {
        let list = [], label = 'الحساب المتأثر';
        const $sel = $('#' + selectId);

        if (!type || type === '') {
            $sel.prop('disabled', true).empty().append('<option value="">-- اختر نوع التوصيل أولاً --</option>');
            $('#' + labelId).text('الحساب المتأثر');
            $('#' + receivedFieldId).hide();
            return;
        }

        $sel.prop('disabled', false);
        if (type === 'cash') {
            list = entitiesData.cashboxes;
            label = 'الحساب: الصناديق';
            $('#' + receivedFieldId).show();
        } else if (type === 'credit') {
            list = entitiesData.customers;
            label = 'الحساب: العملاء';
            $('#' + receivedFieldId).hide();
        } else if (type === 'bank_transfer') {
            list = entitiesData.banks;
            label = 'الحساب: البنوك';
            $('#' + receivedFieldId).hide();
        } else if (type === 'agent') {
            list = entitiesData.agents;
            label = 'الحساب: الوكلاء';
            $('#' + receivedFieldId).hide();
        }

        $('#' + labelId).text(label);
        $sel.empty().append('<option value="">-- اختر --</option>');
        list.forEach(item => {
            const customerId = item.customer_id || '';
            const agentId = item.agent_id || '';
            const displayName = item.display_name || (item.account_code + ' - ' + (item.name || item.account_name_ar));
            $sel.append(`<option value="${item.id}" data-customer-id="${customerId}" data-agent-id="${agentId}">${displayName}</option>`);
        });
        $sel.trigger('change');
    }

    // دالة تحديث المنطق (سعر الصرف)
    function updateLogic() {
        const recordPurchase = $('#record_purchase').val() === '1';
        const purCurrencyId = $('#main_currency_id').val();
        const saleCurrencyId = $('#sale_currency_id').val();
        const purOpt = $('#main_currency_id option:selected');
        const saleOpt = $('#sale_currency_id option:selected');

        if (recordPurchase) {
            $('#main_currency_label').html('عملة التكلفة (المورد) <span class="text-danger">*</span>');
        } else {
            $('#main_currency_label').html('العملة <span class="text-danger">*</span>');
        }

        if (purCurrencyId && saleCurrencyId && purCurrencyId != saleCurrencyId && recordPurchase) {
            $('#exchange_rate_container').show();
            const purSymbol = purOpt.data('symbol') || '---';
            const saleSymbol = saleOpt.data('symbol') || '---';
            const purBuy = parseFloat(purOpt.data('buy')) || 1;
            const saleSell = parseFloat(saleOpt.data('sell')) || 1;
            const rate = purBuy / saleSell;

            $('.pur-symbol').text(purSymbol);
            $('.sale-symbol').text(saleSymbol);
            $('#exchange_rate_label').html('1 ' + purSymbol + ' = ? ' + saleSymbol);
            $('#invoice_exchange_rate').val(rate.toFixed(6));
        } else {
            $('#invoice_exchange_rate').val('1.000000');
            $('#exchange_rate_container').hide();
        }
        calculateEquivalent();
        validateDiscount();
    }

    // دالة حساب التكلفة المعادلة
    function calculateEquivalent() {
        const cost = parseFloat($('#cost_amount').val()) || 0;
        const saleCurrencyId = $('#sale_currency_id').val();
        const mainCurrencyId = $('#main_currency_id').val();
        const rate = parseFloat($('#invoice_exchange_rate').val()) || 1;
        const recordPurchase = $('#record_purchase').val() === '1';
        
        let equivalent = cost;
        if (recordPurchase && saleCurrencyId != mainCurrencyId) {
            equivalent = cost * rate;
        }
        const saleSymbol = $('#sale_currency_id option:selected').data('symbol') || 'ر.ي';
        $('#equivalent_cost_display').val(equivalent.toLocaleString(undefined, { minimumFractionDigits: 2 }) + ' ' + saleSymbol);
    }

    // دالة التحقق من صحة الخصم
    function validateDiscount() {
        const total = parseFloat($('#total_amount').val()) || 0;
        const discount = parseFloat($('#discount').val()) || 0;
        const cost = parseFloat($('#cost_amount').val()) || 0;
        const saleCurrencyId = $('#sale_currency_id').val();
        const mainCurrencyId = $('#main_currency_id').val();
        const rate = parseFloat($('#invoice_exchange_rate').val()) || 1;
        const recordPurchase = $('#record_purchase').val() === '1';

        let costInSaleCurrency = cost;
        if (recordPurchase && saleCurrencyId != mainCurrencyId) {
            costInSaleCurrency = cost * rate;
        }
        const netPrice = total - discount;

        if (discount > 0 && netPrice < costInSaleCurrency - 0.01) {
            $('#discount').addClass('is-invalid');
            const maxAllowed = Math.max(0, total - costInSaleCurrency);
            const errorMsg = `لا يمكن أن يقل السعر الصافي عن التكلفة (${costInSaleCurrency.toFixed(2)}). أقصى خصم مسموح: ${maxAllowed.toFixed(2)}`;
            if (!$('#discount_error').length) {
                $('#discount').after(`<div id="discount_error" class="invalid-feedback small fw-bold">${errorMsg}</div>`);
            } else {
                $('#discount_error').text(errorMsg);
            }
            return false;
        } else {
            $('#discount').removeClass('is-invalid');
            $('#discount_error').remove();
            return true;
        }
    }

    // دالة حساب الضريبة والصافي
    function calculateTaxAndNet() {
        let total = parseFloat($('#total_amount').val()) || 0;
        let discount = parseFloat($('#discount').val()) || 0;
        let afterDiscount = total - discount;
        let taxRate = parseFloat($('#tax_rate').val()) || 0;
        let taxAmount = afterDiscount * (taxRate / 100);
        let netAmount = afterDiscount + taxAmount;
        $('#tax_amount').val(taxAmount.toFixed(2));
        $('#net_amount').val(netAmount.toFixed(2));
    }

                    $('#account_balance_info').removeClass('d-none');
                } else {

                }

        }
    });

                    $('#supplier_unified_limit_display').text(debitLimitBase > 0 ? debitLimitBase.toLocaleString(undefined, {minimumFractionDigits: 2}) + ' ' + baseCurrencySymbol : 'غير محدد');

        const serviceName = $(this).val();
        updateServiceAccounts(serviceName);
        if (serviceName && serviceName !== 'general') {
            const serviceId = $(this).find(':selected').data('id');
            $.ajax({
                url: 'ajax_get_service_price.php',

                        $('#cost_amount').val(res.purchase_price);

                            $('#sale_currency_id').val(res.currency_id).trigger('change');
                            $('#main_currency_id').val(res.currency_id).trigger('change');
                        }
                        if (res.supplier_id) {
                            $('#supplier_select').val(res.supplier_id).trigger('change');
                        }
                        if (res.customer_id || res.agent_id) {
                            const type = res.customer_id ? 'credit' : 'agent';
                            const targetId = res.customer_id || res.agent_id;
                            const dataAttr = res.customer_id ? 'customer-id' : 'agent-id';

                                    $('#account_select').val(optionToSelect.val()).trigger('change');
                                }
                            }, 200);
                        }
                        updateLogic();
                    }
                }
            });
        }
    });
